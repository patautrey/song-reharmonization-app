function playChordSequence(isReharm) { const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); const chords = sectionData[currentSection]; const instrument = document.getElementById('instrumentSelect').value; let time = audioCtx.currentTime; const noteMap = { 'C': 261.63, 'C#': 277.18, 'Db': 277.18, 'D': 293.66, 'D#': 311.13, 'Eb': 311.13, 'E': 329.63, 'F': 349.23, 'F#': 369.99, 'Gb': 369.99, 'G': 392.00, 'G#': 415.30, 'Ab': 415.30, 'A': 440.00, 'A#': 466.16, 'Bb': 466.16, 'B': 493.88 }; chords.forEach((c, idx) => { let chordStr = isReharm ? c.reharm : c.base; if (!chordStr || chordStr === '-' || chordStr.includes('Shell') || chordStr.includes('No Root') || chordStr.includes('Pivot') || chordStr.includes('Prep') || chordStr.includes('Direct')) return; let rootMatch = chordStr.match(/^[A-G][b#]*/); let root = rootMatch ? rootMatch[0] : 'C'; let freq = noteMap[root] || 261.63; let frequencies = [freq, freq * 1.25, freq * 1.5]; if(chordStr.includes('m')) { frequencies[1] = freq * 1.2; } frequencies.forEach(f => { let osc = audioCtx.createOscillator(); let gain = audioCtx.createGain(); if (instrument === 'Acoustic Guitar') { osc.type = 'sawtooth'; gain.gain.setValueAtTime(0.08, time + (idx * 0.5)); gain.gain.exponentialRampToValueAtTime(0.0001, time + (idx * 0.5) + 0.35); } else { osc.type = 'triangle'; gain.gain.setValueAtTime(0.12, time + (idx * 0.5)); gain.gain.exponentialRampToValueAtTime(0.0001, time + (idx * 0.5) + 0.5); } osc.frequency.setValueAtTime(f, time + (idx * 0.5)); osc.connect(gain); gain.connect(audioCtx.destination); osc.start(time + (idx * 0.5)); osc.stop(time + (idx * 0.5) + 0.6); }); }); document.getElementById('audioStatusText').textContent = `Playing [${instrument}] ${isReharm ? 'Reharmonized' : 'Base'} progression for ${currentSection}...`; } function playBase() { playChordSequence(false); } function playReharm() { playChordSequence(true); }
